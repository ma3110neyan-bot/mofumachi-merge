受領run `font-only-20261104-143729-65167335` は補助文字の行高超過で1/2失敗。今回の再確認は、[固定行の局所修正と既存フォルダを再利用する手順](font-only-header-fix-2026-11-04.md)を使用する。以下は新規に撮影用フォルダを作る場合の手順。

Unity Editorを閉じ、Windows PowerShell 5.1に以下をそのまま貼り付ける。専用の子PowerShellだけに`RemoteSigned`を適用して実行する。元の作業フォルダの未コミット差分・`.utmp`・stash・過去の撮影結果は保持し、別の新しいクリーンworktreeで撮影する。失敗した場合は再実行せず、そのrunを保持する。

```powershell
powershell.exe -NoProfile -ExecutionPolicy RemoteSigned -Command {
    $ErrorActionPreference = 'Stop'
    $fontSource = 'C:\Users\User\Desktop\Mofumachi\mofumachi-merge'
    Set-Location -LiteralPath $fontSource
    . '.\scripts\build-android.ps1'
    $fontGit = Find-SourceGit
    $fontBranch = 'feat/daily-world-pass-20261016'
    $currentBranch = & $fontGit branch --show-current
    if ($LASTEXITCODE -ne 0 -or $currentBranch.Trim() -ne $fontBranch) { throw 'Unexpected source branch.' }
    & $fontGit fetch origin $fontBranch
    if ($LASTEXITCODE -ne 0) { throw 'Fetch failed. No capture started.' }
    $fontRevision = (& $fontGit rev-parse ('refs/remotes/origin/' + $fontBranch)).Trim()
    if ($LASTEXITCODE -ne 0) { throw 'Cannot resolve font candidate.' }
    & $fontGit merge-base --is-ancestor 7d95057 $fontRevision
    if ($LASTEXITCODE -ne 0) { throw 'Expected font baseline missing.' }
    $fontTree = Join-Path (Split-Path -Parent $fontSource) ('font-' + [guid]::NewGuid().ToString('N').Substring(0,8))
    & $fontGit worktree add --detach $fontTree $fontRevision
    if ($LASTEXITCODE -ne 0) { throw 'Worktree creation failed.' }
    Set-Location -LiteralPath $fontTree
    $fontStatus = @(& $fontGit status --porcelain=v1 --untracked-files=all)
    if ($LASTEXITCODE -ne 0 -or $fontStatus.Count -ne 0) { throw 'Worktree is not clean. No capture started.' }
    Write-Host ('Font-only candidate: ' + $fontRevision)
    & '.\scripts\prepare-font-only-preview.ps1'
}
```
