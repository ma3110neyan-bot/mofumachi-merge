using System;using System.Linq;using Mofumachi.Presentation;using Mofumachi.Core;using Mofumachi.Tests;
class Program{static void Main(string[] args){var model=new TownWorldWalkModel(0,"momo");model.Restore("momo",CharacterAuthoredCaptureRoute.MomoStart);model.Restore("noah",CharacterAuthoredCaptureRoute.PartnerStart);var d=NoahAuthoredDirection.Left;var observed=new System.Collections.Generic.HashSet<NoahAuthoredDirection>();
foreach(var target in CharacterAuthoredCaptureRoute.Targets){if(!model.TryGuidePoint(target))throw new Exception("Guide rejected "+target.X+","+target.Y);for(int i=0;i<180;i++){foreach(var r in model.Residents.Where(r=>!r.Controlled))model.Pause(r.Id,120);model.Step(1f/30);if(model.Hero.IsWalking){d=NoahAuthoredMotion.Direction(model.Hero.Position.X-model.Hero.Previous.X,model.Hero.Position.Y-model.Hero.Previous.Y,d);observed.Add(d);}if(i>12&&!model.Hero.IsWalking)break;}Console.WriteLine($"target({target.X},{target.Y}), actual({model.Hero.Position.X},{model.Hero.Position.Y}), facing {d}, walking {model.Hero.IsWalking}");if(model.Hero.IsWalking)throw new Exception("Route did not settle");model.CancelHero();}
foreach(var required in new[]{NoahAuthoredDirection.Left,NoahAuthoredDirection.Right,NoahAuthoredDirection.Back,NoahAuthoredDirection.Front})if(!observed.Contains(required))throw new Exception("Missing actual movement direction "+required);
if(!model.TryGuidePoint(CharacterAuthoredCaptureRoute.ConversationTarget))throw new Exception("Approach rejected");bool ready=false;
for(int i=0;i<180;i++){foreach(var r in model.Residents.Where(r=>!r.Controlled))model.Pause(r.Id,4);model.Step(1f/30);if(CharacterAuthoredCaptureRoute.ReadyForConversation(model)){ready=true;break;}}
if(!ready)throw new Exception("No lateral conversation approach");Console.WriteLine($"Conversation approach at({model.Hero.Position.X},{model.Hero.Position.Y}), facing Right, intentional stop");model.CancelHero();
Console.WriteLine("Single-character pure model approach passed; not native rendering.");}}
