using System;using System.Linq;using System.Reflection;using System.Collections.Generic;using System.Text.Json;using System.Diagnostics;using System.IO;using Mofumachi.Core;using Mofumachi.Presentation;
class Program {
record Row(int Case,int Tick,string Id,float X,float Y,float Dx,float Dy,float TargetX,float TargetY,string Facing,int Reverse,int Replans);
record Result(int Case,int Degrees,float Offset,bool Hero,int Frames,bool Arrived,int Reverse,int Bursts,int Flips,int YCrossings,int Replans,float MaxAngle,float MinDistance);
static void Main(){var timer=Stopwatch.StartNew();var results=new List<Result>();int cid=0;var guide=typeof(TownWorldWalkModel).GetMethod("Guide",BindingFlags.Instance|BindingFlags.NonPublic);var route=typeof(TownWorldResident).GetField("Route",BindingFlags.Instance|BindingFlags.NonPublic);
foreach(bool heroMode in new[]{false,true})foreach(int degrees in Enumerable.Range(0,24).Select(i=>i*15))foreach(float offset in new[]{-.04f,0,.04f}) {
int caseId=cid++;var m=new TownWorldWalkModel(1,heroMode?"momo":"noah");var a=m.Residents.Single(r=>r.Id=="momo");var b=m.Residents.Single(r=>r.Id=="lulu");float angle=degrees*MathF.PI/180,dx=MathF.Cos(angle),dy=MathF.Sin(angle);
TownPoint P(float along,float across)=>new(dx*along-dy*across,-.30f+dy*along+dx*across);
var startA=P(-.22f,offset);var startB=P(.22f,-offset);var goalA=P(.22f,offset);var goalB=P(-.22f,-offset);
m.Restore(a.Id,startA);m.Restore(b.Id,startB);foreach(var r in m.Residents.Where(r=>r!=a&&r!=b))m.Restore(r.Id,new TownPoint(-1.8f,-.4f));
if(!m.Area.Contains(startA)||!m.Area.Contains(startB)||!m.Area.Contains(goalA)||!m.Area.Contains(goalB))continue;
if(!(bool)guide.Invoke(m,new object[]{a,goalA,null,true,false})||!(bool)guide.Invoke(m,new object[]{b,goalB,null,true,false}))continue;
var tracker=new TownCharacterFacing();var facing=NoahAuthoredDirection.Left;float px=0,py=0,max=0,min=10;int reverses=0,flips=0,bursts=0,crossings=0,replans=0,previousSide=Math.Sign(a.Position.Y-b.Position.Y),i=0;var window=new Queue<int>();var rows=new List<Row>();object previousRoute=route.GetValue(a);
for(;i<1200&&(a.IsWalking||b.IsWalking);i++){
foreach(var r in m.Residents.Where(r=>r!=a&&r!=b||!r.IsWalking))m.Pause(r.Id,4);m.Step(1f/30);float x=a.Position.X-a.Previous.X,y=a.Position.Y-a.Previous.Y,n=MathF.Sqrt(x*x+y*y),prev=MathF.Sqrt(px*px+py*py);int reverse=0;
if(n>.00001f&&prev>.00001f&&a.IsWalking){float dot=Math.Clamp((x*px+y*py)/(n*prev),-1,1);max=Math.Max(max,MathF.Acos(dot)*180/MathF.PI);if(dot<-.1f){reverse=1;reverses++;window.Enqueue(i);}while(window.Count>0&&i-window.Peek()>15)window.Dequeue();bursts=Math.Max(bursts,window.Count);}
if(n>.00001f){px=x;py=y;var next=tracker.Update(facing,x,y,1f/30);if(next!=facing)flips++;facing=next;}
min=Math.Min(min,TownWalkableArea.Distance(a.Position,b.Position));int side=Math.Sign(a.Position.Y-b.Position.Y);if(side!=0&&previousSide!=0&&side!=previousSide)crossings++;if(side!=0)previousSide=side;
var nowRoute=route.GetValue(a);if(!ReferenceEquals(previousRoute,nowRoute)){replans++;previousRoute=nowRoute;}
rows.Add(new(caseId,i,a.Id,a.Position.X,a.Position.Y,x,y,a.Target.X,a.Target.Y,facing.ToString(),reverse,replans));
}
var result=new Result(caseId,degrees,offset,heroMode,i,!a.IsWalking&&!b.IsWalking,reverses,bursts,flips,crossings,replans,max,min);results.Add(result);
if(bursts>=3||!result.Arrived)File.WriteAllText("/workspace/runtime/characters-stability-nov01/probe/trace-"+caseId+".json",JsonSerializer.Serialize(rows));
if(timer.Elapsed.TotalSeconds>45){Console.WriteLine("Diagnostic45s budget reached");break;}
}
File.WriteAllText("/workspace/runtime/characters-stability-nov01/probe/results-before.json",JsonSerializer.Serialize(results));foreach(var r in results.OrderByDescending(r=>r.Bursts).ThenByDescending(r=>r.Reverse).Take(12))Console.WriteLine(JsonSerializer.Serialize(r));Console.WriteLine($"cases={results.Count} seconds={timer.Elapsed.TotalSeconds:F2}");}
}
